slownik_rzymskie = {
    'I': 1, 
    'V': 5, 
    'X': 10, 
    'L': 50, 
    'C': 100, 
    'D': 500, 
    'M': 1000
}

liczba_rzymska = input("Podaj liczbę rzymską: ")

suma = 0

for i in range(len(liczba_rzymska)):
    if i + 1 < len(liczba_rzymska):
        if slownik_rzymskie[liczba_rzymska[i]] >= slownik_rzymskie[liczba_rzymska[i + 1]]:
            suma += slownik_rzymskie[liczba_rzymska[i]]
        else:
            suma -= slownik_rzymskie[liczba_rzymska[i]]
    else:
        suma += slownik_rzymskie[liczba_rzymska[i]]

print("Wartość arabska:", suma)

sprawdzone_litery = ""

for indeks, litera in enumerate(napis):
    if litera not in sprawdzone_litery:
        indeksy = ""
        liczba_wystapien = 0
        
        for i, l in enumerate(napis):
            if l == litera:
                indeksy += str(i) + " "  # Zapisujemy indeks litery
                liczba_wystapien += 1  # Zwiększamy liczbę wystąpień
        
        print(f"{litera}: {indeksy}({liczba_wystapien})")
        
        sprawdzone_litery += litera

slownik_dziesiatki = {
    'dwadzieścia': 20,
    'trzydzieści': 30,
    'czterdzieści': 40,
    'pięćdziesiąt': 50
}

slownik_jednosci = {
    'zero': 0,
    'jeden': 1,
    'dwa': 2,
    'trzy': 3,
    'cztery': 4,
    'pięć': 5,
    'sześć': 6,
    'siedem': 7,
    'osiem': 8,
    'dziewięć': 9
}

def zamien_na_liczbe(napis):
    wyrazy = napis.split()

    dziesiatki = slownik_dziesiatki[wyrazy[0]]

    if len(wyrazy) > 1:
        jednosci = slownik_jednosci[wyrazy[1]]
    else:
        jednosci = 0  # Jeśli nie ma drugiego słowa, jedności to zero

    return dziesiatki + jednosci

napis = input("Podaj liczbę słownie (np. 'dwadzieścia jeden'): ")
liczba = zamien_na_liczbe(napis)
print("Liczba:", liczba)
