from abc import ABC, abstractmethod
import math

# Tworzymy klasę abstrakcyjną Kształt
class Kształt(ABC):
    def __init__(self, nazwa):
        self.__nazwa = nazwa  # Prywatny atrybut z nazwą figury

    @property
    def nazwa(self):
        return self.__nazwa

    @abstractmethod
    def pole(self):
        pass  # Metoda abstrakcyjna do obliczania pola


# Klasa Koło dziedziczy z klasy Kształt
class Kolo(Kształt):
    def __init__(self, nazwa, promien):
        super().__init__(nazwa)
        self.promien = promien  # Promień koła

    def pole(self):
        return math.pi * (self.promien ** 2)  # Wzór na pole koła: πr²


# Klasa Trójkąt dziedziczy z klasy Kształt
class Trojkat(Kształt):
    def __init__(self, nazwa, bok_a, bok_b, bok_c):
        super().__init__(nazwa)
        self.bok_a = bok_a  # Boki trójkąta
        self.bok_b = bok_b
        self.bok_c = bok_c

    def pole(self):
        # Wzór Herona: sqrt(s * (s - a) * (s - b) * (s - c))
        s = (self.bok_a + self.bok_b + self.bok_c) / 2  # Połowa obwodu
        return math.sqrt(s * (s - self.bok_a) * (s - self.bok_b) * (s - self.bok_c))


# Klasa Prostokąt dziedziczy z klasy Kształt
class Prostokat(Kształt):
    def __init__(self, nazwa, bok_a, bok_b):
        super().__init__(nazwa)
        self.bok_a = bok_a  # Boki prostokąta
        self.bok_b = bok_b

    def pole(self):
        return self.bok_a * self.bok_b  # Wzór na pole prostokąta: a * b


# Klasa Kwadrat - wersja foremna Prostokąta
class Kwadrat(Prostokat):
    def __init__(self, nazwa, bok):
        super().__init__(nazwa, bok, bok)  # Bok kwadratu taki sam na każdym boku


# Klasa TrojkatRownoboczny - wersja foremna Trójkąta
class TrojkatRownoboczny(Trojkat):
    def __init__(self, nazwa, bok):
        super().__init__(nazwa, bok, bok, bok)  # Wszystkie boki trójkąta równobocznego są równe


# Funkcja główna main
def main():
    # Tworzymy listę z różnymi figurami
    figury = [
        Kolo("Koło", 5),
        Trojkat("Trójkąt", 3, 4, 5),
        Prostokat("Prostokąt", 4, 6),
        Kwadrat("Kwadrat", 4),
        TrojkatRownoboczny("Trójkąt równoboczny", 5)
    ]

    # Iteracja po figurach i wyświetlanie nazw i pól
    for figura in figury:
        print(f"Figura: {figura.nazwa}, Pole: {figura.pole():.2f}")


# Wywołanie funkcji main
if __name__ == "__main__":
    main()

from abc import ABC, abstractmethod
import math

# Tworzymy klasę abstrakcyjną Kształt
class Kształt(ABC):
    def __init__(self, nazwa):
        self.__nazwa = nazwa  # Prywatny atrybut z nazwą figury

    @property
    def nazwa(self):
        return self.__nazwa

    @abstractmethod
    def pole(self):
        pass  # Metoda abstrakcyjna do obliczania pola


# Klasa Koło dziedziczy z klasy Kształt
class Kolo(Kształt):
    def __init__(self, nazwa, promien):
        super().__init__(nazwa)
        self.promien = promien  # Promień koła

    def pole(self):
        return math.pi * (self.promien ** 2)  # Wzór na pole koła: πr²


# Klasa Trójkąt dziedziczy z klasy Kształt
class Trojkat(Kształt):
    def __init__(self, nazwa, bok_a, bok_b, bok_c):
        super().__init__(nazwa)
        self.bok_a = bok_a  # Boki trójkąta
        self.bok_b = bok_b
        self.bok_c = bok_c

    def pole(self):
        # Wzór Herona: sqrt(s * (s - a) * (s - b) * (s - c))
        s = (self.bok_a + self.bok_b + self.bok_c) / 2  # Połowa obwodu
        return math.sqrt(s * (s - self.bok_a) * (s - self.bok_b) * (s - self.bok_c))


# Klasa Prostokąt dziedziczy z klasy Kształt
class Prostokat(Kształt):
    def __init__(self, nazwa, bok_a, bok_b):
        super().__init__(nazwa)
        self.bok_a = bok_a  # Boki prostokąta
        self.bok_b = bok_b

    def pole(self):
        return self.bok_a * self.bok_b  # Wzór na pole prostokąta: a * b


# Klasa Kwadrat - wersja foremna Prostokąta
class Kwadrat(Prostokat):
    def __init__(self, nazwa, bok):
        super().__init__(nazwa, bok, bok)  # Bok kwadratu taki sam na każdym boku
        self.liczba_bokow = 4  # Kwadrat ma zawsze 4 boki


# Klasa TrojkatRownoboczny - wersja foremna Trójkąta
class TrojkatRownoboczny(Trojkat):
    def __init__(self, nazwa, bok):
        super().__init__(nazwa, bok, bok, bok)  # Wszystkie boki trójkąta równobocznego są równe
        self.liczba_bokow = 3  # Trójkąt równoboczny ma zawsze 3 boki


# Klasa Bryła, która jest nadrzędna dla brył zbudowanych z figur foremnych
class Bryla(Kształt):
    def __init__(self, nazwa, sciany):
        super().__init__(nazwa)
        self.sciany = sciany  # Lista ścian tworzących bryłę

    def pole(self):
        # Suma pól wszystkich ścian bryły
        return sum([sciana.pole() for sciana in self.sciany])


# Klasa Platon - mixin dodający metodę do wyliczania liczby wierzchołków
class Platon:
    def wierzcholki(self):
        liczba_scian = len(self.sciany)
        liczba_bokow_sciany = self.sciany[0].liczba_bokow  # Z boku pierwszej ściany
        return int(2 - liczba_scian + (liczba_scian * liczba_bokow_sciany) / 2)


# Klasa Czworoscian - bryła foremna zbudowana z trójkątów równobocznych
class Czworoscian(Bryla, Platon):
    def __init__(self, nazwa, bok):
        sciany = [TrojkatRownoboczny("Sciana trójkątna", bok) for _ in range(4)]
        super().__init__(nazwa, sciany)


# Klasa Sześcian - bryła foremna zbudowana z kwadratów
class Szescian(Bryla, Platon):
    def __init__(self, nazwa, bok):
        sciany = [Kwadrat("Sciana kwadratowa", bok) for _ in range(6)]
        super().__init__(nazwa, sciany)


# Klasa Piramida - bryła z podstawą kwadratu i czterema ścianami bocznymi (trójkątami)
class Piramida(Bryla):
    def __init__(self, nazwa, bok):
        sciany = [Kwadrat("Podstawa kwadratowa", bok)] + [TrojkatRownoboczny("Sciana trójkątna", bok) for _ in range(4)]
        super().__init__(nazwa, sciany)


# Funkcja główna main
def main():
    # Tworzymy listę brył
    bryly = [
        Czworoscian("Czworościan", 5),
        Szescian("Sześcian", 4),
        Piramida("Piramida", 6)
    ]

    # Iteracja po bryłach i wyświetlanie ich pól, a dla brył platońskich również liczby wierzchołków
    for bryla in bryly:
        print(f"Bryła: {bryla.nazwa}, Pole: {bryla.pole():.2f}")
        if isinstance(bryla, Platon):
            print(f"Liczba wierzchołków: {bryla.wierzcholki()}")


# Wywołanie funkcji main
if __name__ == "__main__":
    main()
